export type PlaceOptions = { detail: ModelDetail; quality?: StageQuality };
export type PlaceModel = {
  group: THREE.Group;
  landmarks: readonly THREE.Vector3[];
  update(cues: StageCues): void;
  dispose(): void;
};
type V3 = [number, number, number];
type Motion = (time: number, cues: StageCues) => void;

function strut(
  k: StageKit,
  mat: 'chrome' | 'paint' | 'wood' | 'glow',
  colour: string,
  a: V3,
  b: V3,
  r: number,
) {
  const p = new THREE.Vector3(...a),
    q = new THREE.Vector3(...b),
    d = q.clone().sub(p);
  const g = new THREE.CylinderGeometry(
    r,
    r,
    d.length(),
    k.quality === 'high' ? 12 : 6,
  );
  g.applyQuaternion(
    new THREE.Quaternion().setFromUnitVectors(
      new THREE.Vector3(0, 1, 0),
      d.normalize(),
    ),
  );
  k.add(mat, colour, g, p.add(q).multiplyScalar(0.5).toArray() as V3);
}

function lamp(k: StageKit, x: number, z: number, h = 3.8, tint = '#ffd39b') {
  k.cyl('paint', '#34464f', 0.043, 0.072, h, [x, h / 2, z]);
  k.cyl('chrome', '#b6c0ad', 0.18, 0.24, 0.18, [x, 0.08, z]);
  k.cyl('glow', tint, 0.15, 0.1, 0.33, [x, h, z], { seg: 8 });
  k.cyl('paint', '#34464f', 0, 0.29, 0.17, [x, h + 0.25, z], { seg: 8 });
  for (const side of [-1, 1])
    k.box(
      'paint',
      '#34464f',
      [0.025, 0.34, 0.025],
      [x + side * 0.13, h, z + 0.08],
    );
}

function sedan(k: StageKit, at: V3, colour: string, yaw = 0, scale = 1) {
  const car = new StageKit(k.quality);
  car.rbox('gloss', colour, [1.85, 0.36, 0.79], 0.16, [0, 0.34, 0]);
  car.rbox('gloss', '#c1c5b5', [0.91, 0.39, 0.72], 0.17, [0.03, 0.63, 0]);
  car.add(
    'glass',
    '#b1d4d5',
    panel(0.52, 0.26, 0.08, 0.025, k.quality),
    [0.03, 0.65, 0.369],
  );
  car.rbox('chrome', '#b7c4cc', [0.08, 0.07, 0.82], 0.025, [-0.93, 0.27, 0]);
  for (const x of [-0.59, 0.56])
    for (const s of [-1, 1]) {
      car.cyl('paint', '#26333f', 0.19, 0.19, 0.07, [x, 0.19, s * 0.39], {
        axis: 'z',
        seg: 12,
      });
      car.cyl('chrome', '#c4d2d5', 0.1, 0.1, 0.08, [x, 0.19, s * 0.39], {
        axis: 'z',
        seg: 10,
      });
      car.box('glow', '#ffe1a4', [0.035, 0.08, 0.095], [-0.92, 0.4, s * 0.28]);
      car.box('glow', '#e26656', [0.03, 0.06, 0.08], [0.92, 0.4, s * 0.28]);
    }
  const parts = car.build();
  for (const [mat, g] of Object.entries(parts)) {
    g.scale(scale, scale, scale);
    g.rotateY(yaw);
    g.translate(...at);
    (k.parts[mat as keyof typeof k.parts] ??= []).push(g);
  }
}

/** Carved carousel pony: connected body/neck/head and two posed legs, not beads. */
function carouselHorse(
  k: StageKit,
  at: V3,
  yaw: number,
  scale: number,
  coat: string,
) {
  const horse = new StageKit(k.quality),
    profile = new THREE.Shape();
  profile.moveTo(-0.4, 0.12);
  profile.bezierCurveTo(-0.5, 0.08, -0.48, -0.11, -0.35, -0.14);
  profile.lineTo(-0.44, -0.37);
  profile.lineTo(-0.3, -0.4);
  profile.lineTo(-0.2, -0.15);
  profile.quadraticCurveTo(-0.03, -0.21, 0.13, -0.12);
  profile.lineTo(0.33, -0.36);
  profile.lineTo(0.45, -0.35);
  profile.lineTo(0.3, -0.07);
  profile.quadraticCurveTo(0.31, 0.1, 0.33, 0.21);
  profile.lineTo(0.49, 0.38);
  profile.lineTo(0.64, 0.35);
  profile.quadraticCurveTo(0.77, 0.36, 0.76, 0.49);
  profile.lineTo(0.51, 0.6);
  profile.lineTo(0.46, 0.73);
  profile.lineTo(0.4, 0.59);
  profile.bezierCurveTo(0.2, 0.56, 0.25, 0.28, 0.06, 0.17);
  profile.lineTo(-0.3, 0.19);
  profile.quadraticCurveTo(-0.37, 0.19, -0.4, 0.12);
  const geo = new THREE.ExtrudeGeometry(profile, {
    depth: 0.17,
    bevelEnabled: true,
    bevelSize: 0.022,
    bevelThickness: 0.025,
    bevelSegments: k.quality === 'high' ? 3 : 2,
    curveSegments: k.quality === 'high' ? 8 : 4,
  });
  geo.translate(0, 0, -0.085);
  horse.add('gloss', coat, geo, [0, 0, 0]);
  horse.rbox('gloss', '#9c4567', [0.26, 0.05, 0.25], 0.025, [-0.03, 0.18, 0]);
  horse.tube(
    'gloss',
    '#735574',
    [
      [-0.4, 0.12, 0],
      [-0.58, 0.02, 0],
      [-0.52, -0.19, 0],
    ],
    0.036,
  );
  horse.tube(
    'gloss',
    '#735574',
    [
      [0.08, 0.19, 0],
      [0.23, 0.34, 0],
      [0.27, 0.51, 0],
      [0.4, 0.58, 0],
    ],
    0.03,
  );
  for (const side of [-1, 1]) {
    horse.sphere('paint', '#334452', 0.024, [0.5, 0.5, side * 0.115], {
      seg: 8,
    });
    horse.tube(
      'chrome',
      '#d1aa79',
      [
        [0.65, 0.44, side * 0.114],
        [0.29, 0.27, side * 0.123],
        [0.06, 0.17, side * 0.135],
      ],
      0.009,
    );
  }
  for (const [mat, part] of Object.entries(horse.build())) {
    part.scale(scale, scale, scale);
    part.rotateY(yaw);
    part.translate(...at);
    (k.parts[mat as keyof typeof k.parts] ??= []).push(part);
  }
}

function pump(k: StageKit, x: number, z: number) {
  k.rbox('chrome', '#acbcc7', [0.84, 0.09, 0.65], 0.045, [x, 0.03, z]);
  k.rbox('gloss', '#bb4f46', [0.67, 1.14, 0.51], 0.13, [x, 0.64, z]);
  k.rbox('gloss', '#eddabb', [0.7, 0.6, 0.52], 0.1, [x, 1.46, z]);
  k.box('paint', '#243945', [0.46, 0.17, 0.035], [x, 1.55, z + 0.27]);
  for (let i = 0; i < 4; i++)
    k.box(
      'glow',
      '#bdedc7',
      [0.042, 0.06, 0.014],
      [x - 0.13 + i * 0.087, 1.55, z + 0.29],
    );
  k.sphere('glow', '#ffe3af', 0.21, [x, 1.91, z], {
    scale: [1, 1.0, 0.47],
    seg: 16,
  });
  k.tube(
    'paint',
    '#243039',
    [
      [x + 0.35, 1.55, z],
      [x + 0.6, 1.31, z],
      [x + 0.59, 0.42, z],
      [x + 0.35, 0.68, z],
    ],
    0.027,
  );
  k.rbox('chrome', '#9eaab3', [0.09, 0.24, 0.07], 0.023, [x + 0.39, 1.25, z]);
}

function tent(k: StageKit, at: V3, radius: number, colour: string, top = 2.8) {
  const n = 12;
  for (let i = 0; i < n; i++) {
    const g = new THREE.ConeGeometry(
      radius,
      top * 0.44,
      1,
      1,
      true,
      (i / n) * Math.PI * 2,
      (Math.PI * 2) / n,
    );
    k.add('gloss', i % 2 ? '#eedeb7' : colour, g, [
      at[0],
      at[1] + top * 0.78,
      at[2],
    ]);
    const a = (i / n) * Math.PI * 2;
    strut(
      k,
      'chrome',
      '#839ab0',
      [at[0] + Math.sin(a) * radius, at[1], at[2] + Math.cos(a) * radius],
      [
        at[0] + Math.sin(a) * radius,
        at[1] + top * 0.57,
        at[2] + Math.cos(a) * radius,
      ],
      0.04,
    );
    k.sphere(
      'glow',
      '#ffd89c',
      0.07,
      [
        at[0] + Math.sin(a) * radius,
        at[1] + top * 0.57,
        at[2] + Math.cos(a) * radius,
      ],
      { seg: 6 },
    );
  }
  k.cyl('gloss', colour, 0.08, 0.11, 0.36, [at[0], at[1] + top + 0.08, at[2]], {
    seg: 8,
  });
}

function shop(
  k: StageKit,
  x: number,
  z: number,
  w: number,
  h: number,
  colour: string,
) {
  k.rbox('paint', colour, [w, h, 2.1], 0.04, [x, h / 2, z]);
  k.box('paint', '#465064', [w + 0.28, 0.19, 2.36], [x, h + 0.04, z]);
  k.box('ceramic', '#bbac91', [w + 0.14, 0.13, 0.14], [x, h - 0.23, z + 1.1]);
  for (const dx of [-w * 0.26, w * 0.26]) {
    k.add('wood', '#594b4c', panel(w * 0.36, 1.3, 0.16, 0.11, k.quality), [
      x + dx,
      1.0,
      z + 1.11,
    ]);
    k.add('glow', '#ddab75', panel(w * 0.29, 1.15, 0.12, 0.03, k.quality), [
      x + dx,
      1.0,
      z + 1.19,
    ]);
    k.box('wood', '#73685d', [0.045, 1.18, 0.04], [x + dx, 1.0, z + 1.22]);
    if (h > 3) {
      k.add('glow', '#a5b1b8', panel(w * 0.22, 0.83, 0.06, 0.03, k.quality), [
        x + dx,
        h - 1.05,
        z + 1.075,
      ]);
      k.box(
        'ceramic',
        '#ac9c8f',
        [w * 0.27, 0.1, 0.18],
        [x + dx, h - 1.52, z + 1.13],
      );
    }
  }
  for (let i = 0; i < 11; i++)
    k.box(
      'gloss',
      i % 2 ? '#edd7ad' : '#517f77',
      [w / 11, 0.075, 0.69],
      [x - w / 2 + ((i + 0.5) * w) / 11, 1.9, z + 1.32],
      { rot: [0.15, 0, 0] },
    );
  if (k.quality === 'high')
    for (let row = 0; row < 12; row++)
      for (let i = 0; i < 5; i++) {
        const dx = -w / 2 + 0.16 + (i * w) / 5 + (row % 2) * 0.1;
        k.box(
          'paint',
          '#977266',
          [w / 5 - 0.05, 0.014, 0.009],
          [x + dx, 0.18 + (row * h) / 12, z + 1.055],
        );
      }
}

function mesa(k: StageKit, at: V3, r: number, h: number, colour: string) {
  const g = new THREE.CylinderGeometry(r * 0.61, r, h, 9, 4);
  const pos = g.getAttribute('position');
  for (let i = 0; i < pos.count; i++) {
    const a = Math.atan2(pos.getZ(i), pos.getX(i));
    const f = 1 + Math.sin(a * 3 + at[0]) * 0.14;
    pos.setXYZ(i, pos.getX(i) * f, pos.getY(i), pos.getZ(i) * f);
  }
  g.computeVertexNormals();
  k.add('paint', colour, g, [at[0], at[1] + h / 2, at[2]]);
  k.cyl(
    'paint',
    '#866b87',
    r * 0.66,
    r * 0.66,
    h * 0.04,
    [at[0], at[1] + h * 0.92, at[2]],
    { seg: 9 },
  );
}

function boat(k: StageKit, colour: string) {
  // Cross-section swept in rings: full sheer and pointed bow, not a cuboid raft.
  const shape = new THREE.Shape();
  shape.moveTo(-1.3, 0.23);
  shape.quadraticCurveTo(-0.7, -0.21, 0.8, -0.17);
  shape.quadraticCurveTo(1.4, 0, 1.55, 0.45);
  shape.lineTo(-1.3, 0.38);
  shape.closePath();
  const g = new THREE.ExtrudeGeometry(shape, {
    depth: 0.73,
    bevelEnabled: true,
    bevelSize: 0.07,
    bevelThickness: 0.08,
    bevelSegments: 2,
  });
  g.translate(0, 0, -0.365);
  k.add('gloss', colour, g, [0, 0.24, 0]);
  k.rbox('wood', '#a78266', [2.18, 0.055, 0.6], 0.04, [0, 0.62, 0]);
  k.rbox('gloss', '#e9d8b8', [0.74, 0.62, 0.56], 0.07, [-0.35, 0.95, 0]);
  k.box('glass', '#97c3c7', [0.38, 0.31, 0.035], [-0.35, 1.05, 0.3]);
  k.box('paint', '#31454c', [0.88, 0.07, 0.69], [-0.35, 1.32, 0]);
  strut(k, 'wood', '#8f8c7d', [0.18, 0.64, 0], [0.18, 2.5, 0], 0.026);
  strut(k, 'paint', '#55626a', [-1.07, 0.58, 0], [0.18, 2.45, 0], 0.008);
  k.box('glow', '#ecc692', [0.09, 0.08, 0.12], [0.18, 2.47, 0]);
}

export function buildPlace(
  materials: Materials,
  place: PlaceId,
  options: PlaceOptions,
): PlaceModel {
  const stage = options.detail === 'stage',
    quality = options.quality ?? (stage ? 'high' : 'low');
  const group = new THREE.Group();
  group.name = `place-${place}-${options.detail}`;
  const own = glowMaterials(materials),
    mats = own.materials,
    k = new StageKit(quality);
  const motions: Motion[] = [],
    owned: { dispose(): void }[] = [own.glow],
    landmarks: THREE.Vector3[] = [];
  // Map vignettes use their own compact composition; no ground tile or diner copy.
  const dynamic = (name: string, source: StageKit, position: V3) => {
    const g = joinKit(group, source, mats, name);
    g.position.set(...position);
    g.userData.dynamic = true;
    return g;
  };
  if (stage) {
    const ground = new THREE.PlaneGeometry(180, 180, 1, 1);
    ground.rotateX(-Math.PI / 2);
    k.add(
      'paint',
      {
        truckstop: '#454455',
        drivein: '#354453',
        fairground: '#49525c',
        harbour: '#234755',
        oldtown: '#525062',
        moonmile: '#66536f',
      }[place],
      ground,
      [5.5, -0.24, -18],
    );
    // Quiet paving/shore approach below the unfolded patio, extending to screen edges.
    if (place !== 'harbour' && place !== 'moonmile')
      k.box('paint', '#6b6470', [60, 0.015, 5.3], [5.5, -0.22, 13.65]);
  }

  if (place === 'truckstop') {
    const h = stage ? 3.65 : 4.5,
      sign: V3 = stage ? [3.3, 0, -3.25] : [0, 0, 0];
    k.cyl('chrome', '#9facb2', 0.13, 0.2, h, [sign[0], h / 2, sign[2]]);
    k.add(
      'gloss',
      '#a9623f',
      panel(stage ? 3.8 : 2.6, stage ? 1.12 : 1, 0.38, 0.17, quality),
      [sign[0], h, sign[2]],
    );
    neon(
      k,
      'FUEL',
      stage ? 0.65 : 0.46,
      [sign[0], h - (stage ? 0.29 : 0.2), sign[2] + 0.12],
      '#ffd084',
    );
    k.rbox('glow', '#f4ae65', [stage ? 4.1 : 2.85, 0.06, 0.08], 0.025, [
      sign[0],
      h - 0.66,
      sign[2] + 0.07,
    ]);
    const px = stage ? -4.9 : -1.4,
      pz = stage ? -3.6 : 1.0,
      s = stage ? 1 : 0.6;
    const pk = new StageKit(quality);
    for (const x of [-1.2, 1.2]) pump(pk, x, 0);
    for (const x of [-2.1, 2.1])
      pk.cyl('chrome', '#b3c4c6', 0.07, 0.07, 2.5, [x, 1.25, 0]);
    pk.rbox('gloss', '#dfbf91', [5.0, 0.25, 2.45], 0.1, [0, 2.5, 0]);
    pk.box('glow', '#ffc588', [4.8, 0.055, 0.06], [0, 2.48, 1.24]);
    const p = joinKit(group, pk, mats, 'pumps');
    p.position.set(px, 0, pz);
    p.scale.setScalar(s);
    if (stage) {
      for (let n = 0; n < 3; n++) {
        const x = 13.9 + n * 5.7,
          z = -5.9 - (n % 2) * 3.1;
        k.rbox(
          'paint',
          ['#869da6', '#acada2', '#735d66'][n],
          [3.4, 2.7, 7.6],
          0.17,
          [x, 1.43, z],
        );
        k.box('gloss', '#a45742', [3.4, 0.26, 0.03], [x, 1.5, z + 3.82]);
        for (let i = 0; i < 15; i++)
          k.box(
            'chrome',
            '#8ca1b2',
            [0.018, 2.2, 0.018],
            [x - 1.5 + i * 0.215, 1.48, z + 3.815],
          );
        for (const dx of [-1.05, 1.05])
          k.cyl('paint', '#273342', 0.4, 0.4, 0.24, [x + dx, 0.16, z + 2.8], {
            axis: 'x',
            seg: 12,
          });
      }
      k.box('paint', '#2e3444', [110, 0.02, 4.4], [0, -0.215, -13]);
      for (let i = -18; i < 22; i++)
        k.box('paint', '#aa9c88', [1.15, 0.008, 0.065], [i * 3.2, -0.2, -13]);
      const tk = new StageKit(quality);
      sedan(tk, [0, 0, 0], '#c28c57');
      const traffic = dynamic('highway-traffic', tk, [0, -0.19, -12.7]);
      motions.push((t) => {
        traffic.position.x = -32 + ((t * 3.2) % 65);
      });
      lamp(k, -11, -5, 6.4, '#f7b278');
      lamp(k, 23, -5, 6.4, '#f7b278');
    }
    landmarks.push(new THREE.Vector3(sign[0], h + 0.8, sign[2]));
  } else if (place === 'drivein') {
    const center = stage ? ([5.5, 3.0, -3.65] as V3) : ([0, 2.0, 0] as V3);
    const w = stage ? 10.0 : 4.5,
      h = stage ? 3.0 : 2.5;
    k.rbox('paint', '#344964', [w + 0.48, h + 0.5, 0.48], 0.16, center);
    k.rbox('chrome', '#a5c2d2', [w + 0.13, h + 0.13, 0.07], 0.03, [
      center[0],
      center[1],
      center[2] + 0.27,
    ]);
    for (const s of [-1, 1])
      strut(
        k,
        'paint',
        '#627582',
        [center[0] + s * w * 0.42, 0, center[2]],
        [center[0] + s * w * 0.42, center[1] + h * 0.4, center[2]],
        0.14,
      );
    const screenMat = new THREE.MeshBasicMaterial({
      color: '#b2d5dd',
      toneMapped: false,
    });
    owned.push(screenMat);
    const screenColour = screenMat.color.clone();
    const screenGeo = panel(w - 0.18, h - 0.16, 0.08, 0.02, quality);
    const screen = new THREE.Mesh(screenGeo, screenMat);
    screen.position.set(center[0], center[1], center[2] + 0.32);
    group.add(screen);
    // Film is a considered physical silhouette of a rocket above layered mountains.
    const film = new StageKit(quality),
      f = stage ? 0.66 : 0.43;
    film.sphere(
      'glow',
      '#ffdfa1',
      0.62 * f,
      [center[0] - 2.9 * f, center[1] + 1.55 * f, center[2] + 0.35],
      { scale: [1, 1, 0.035], seg: 24 },
    );
    for (let n = 0; n < 4; n++) {
      const s = new THREE.Shape();
      s.moveTo(-1.7 * f, 0);
      s.lineTo(-0.6 * f, (0.5 + (n % 2)) * f);
      s.lineTo(0.3 * f, 0.7 * f);
      s.lineTo(1.7 * f, 0);
      s.closePath();
      film.add(
        'glow',
        n % 2 ? '#3e6b8d' : '#527fa0',
        new THREE.ShapeGeometry(s),
        [
          center[0] + (n - 1.5) * 2.3 * f,
          center[1] - h * 0.44,
          center[2] + 0.355,
        ],
      );
    }
    film.capsule(
      'glow',
      '#e9ebd3',
      0.2 * f,
      1.15 * f,
      [center[0] + 1.5 * f, center[1] + 0.3 * f, center[2] + 0.38],
      { rot: [0, 0, -0.4], scale: [1, 1, 0.14] },
    );
    film.tube(
      'glow',
      '#eaaa75',
      [
        [center[0] + 1.22 * f, center[1] - 0.42 * f, center[2] + 0.39],
        [center[0] + 0.58 * f, center[1] - 1.15 * f, center[2] + 0.39],
      ],
      0.1 * f,
    );
    joinKit(group, film, mats, 'movie');
    motions.push((t, cues) => {
      screenMat.color
        .copy(screenColour)
        .multiplyScalar(lightLevel(cues) * (0.82 + Math.sin(t * 11) * 0.014));
    });
    if (stage) {
      for (let row = 0; row < 2; row++)
        for (let i = 0; i < 7; i++) {
          const px = -7.5 + i * 4.5;
          // Rows flank the trailer; a parked car never occupies its closed back.
          if (px > -2 && px < 13) continue;
          sedan(
            k,
            [px, -0.24, -2.8 - row * 2.8],
            ['#b36375', '#a9b9a3', '#719bad', '#bba17d'][i % 4],
            Math.PI / 2,
            1.13,
          );
        }
      k.rbox('paint', '#526479', [2.2, 1.8, 1.8], 0.15, [-7.2, 0.9, 0.8]);
