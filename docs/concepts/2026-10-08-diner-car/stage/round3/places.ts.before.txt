function strut(
  k: StageKit,
  mat: 'chrome' | 'paint' | 'wood' | 'glow',
  colour: string,
  a: V3,
  b: V3,
  r: number,
) {
  const p = new THREE.Vector3(...a),
    q = new THREE.Vector3(...b),
    d = q.clone().sub(p);
  const g = new THREE.CylinderGeometry(
    r,
    r,
    d.length(),
    k.quality === 'high' ? 12 : 6,
  );
  g.applyQuaternion(
    new THREE.Quaternion().setFromUnitVectors(
      new THREE.Vector3(0, 1, 0),
      d.normalize(),
    ),
  );
  k.add(mat, colour, g, p.add(q).multiplyScalar(0.5).toArray() as V3);
}

function tent(k: StageKit, at: V3, radius: number, colour: string, top = 2.8) {
  const n = 12;
  for (let i = 0; i < n; i++) {
    const g = new THREE.ConeGeometry(
      radius,
      top * 0.44,
      1,
      1,
      true,
      (i / n) * Math.PI * 2,
      (Math.PI * 2) / n,
    );
    k.add('gloss', i % 2 ? '#eedeb7' : colour, g, [
      at[0],
      at[1] + top * 0.78,
      at[2],
    ]);
    const a = (i / n) * Math.PI * 2;
    strut(
      k,
      'chrome',
      '#839ab0',
      [at[0] + Math.sin(a) * radius, at[1], at[2] + Math.cos(a) * radius],
      [
        at[0] + Math.sin(a) * radius,
        at[1] + top * 0.57,
        at[2] + Math.cos(a) * radius,
      ],
      0.04,
    );
    k.sphere(
      'glow',
      '#ffd89c',
      0.07,
      [
        at[0] + Math.sin(a) * radius,
        at[1] + top * 0.57,
        at[2] + Math.cos(a) * radius,
      ],
      { seg: 6 },
    );
  }
  k.cyl('gloss', colour, 0.08, 0.11, 0.36, [at[0], at[1] + top + 0.08, at[2]], {
    seg: 8,
  });
}

function mesa(k: StageKit, at: V3, r: number, h: number, colour: string) {
  const g = new THREE.CylinderGeometry(r * 0.61, r, h, 9, 4);
  const pos = g.getAttribute('position');
  for (let i = 0; i < pos.count; i++) {
    const a = Math.atan2(pos.getZ(i), pos.getX(i));
    const f = 1 + Math.sin(a * 3 + at[0]) * 0.14;
    pos.setXYZ(i, pos.getX(i) * f, pos.getY(i), pos.getZ(i) * f);
  }
  g.computeVertexNormals();
  k.add('paint', colour, g, [at[0], at[1] + h / 2, at[2]]);
  k.cyl(
    'paint',
    '#866b87',
    r * 0.66,
    r * 0.66,
    h * 0.04,
    [at[0], at[1] + h * 0.92, at[2]],
    { seg: 9 },
  );
}

const hash = (x: number, y: number, seed = 0) => {
  let n = Math.imul(x + 619 + seed, 374761393) ^ Math.imul(y + 149, 668265263);
  n = Math.imul(n ^ (n >>> 13), 1274126177);
  return ((n ^ (n >>> 16)) >>> 0) / 4294967295;
};

/** Authored metre-scale ground: different construction, not one noisy paint skin. */
function groundSurface(
  group: THREE.Group,
  place: PlaceId,
  quality: StageQuality,
  owned: { dispose(): void }[],
  motions: Motion[],
) {
  const n = quality === 'high' ? 256 : 128,
    bytes = new Uint8Array(n * n * 4),
    normal = new Uint8Array(n * n * 4),
    heights = new Float32Array(n * n);
  const base = new THREE.Color(
    {
      truckstop: '#3c3a45',
      drivein: '#3b4354',
      fairground: '#34443c',
      harbour: '#163c4b',
      oldtown: '#64616e',
      moonmile: '#554460',
    }[place],
  );
  // THREE.Color is linear; the bitmap has its own sRGB encoding.
  const srgb = base.clone().convertLinearToSRGB();
  const rgb = [srgb.r, srgb.g, srgb.b].map((c) => Math.round(c * 255));
  for (let y = 0; y < n; y++)
    for (let x = 0; x < n; x++) {
      const u = x / n,
        v = y / n,
        grit = hash(x, y, 3),
        broad = hash(x >> 3, y >> 3, 11),
        island =
          Math.sin(u * Math.PI * 6 + Math.sin(v * Math.PI * 4)) *
          Math.cos(v * Math.PI * 6);
      let shade = 0.93 + (grit - 0.5) * 0.09 + island * 0.035,
        h = (grit - 0.5) * 0.02;
      if (place === 'truckstop') {
        const crack =
          Math.abs(Math.sin(u * Math.PI * 8 + Math.sin(v * 29) * 0.6)) < 0.018;
        shade -= crack ? 0.27 : 0;
        shade -= broad < 0.1 ? 0.03 : 0;
      } else if (place === 'drivein') {
        const gravel = hash(x >> 1, y >> 1, 29);
        shade += (gravel - 0.5) * 0.23;
        h += gravel * 0.08;
      } else if (place === 'fairground') {
        const worn = Math.pow(
          Math.max(0, Math.sin(u * 11 + Math.sin(v * 9) * 1.2)),
          4,
        );
        shade += worn * 0.2 + (broad - 0.5) * 0.12;
        h += grit * 0.05;
      } else if (place === 'harbour') {
        const ripple = Math.sin(
          v * Math.PI * 14 + Math.sin(u * Math.PI * 4) * 1.1,
        );
        shade += Math.pow(Math.max(0, ripple), 7) * 0.19;
        h += ripple * 0.12;
      } else if (place === 'oldtown') {
        const row = Math.floor(v * 20),
          a = (u * 13 + (row % 2) * 0.5) % 1,
          b = (v * 20) % 1,
          mortar = a < 0.07 || a > 0.94 || b < 0.09 || b > 0.91;
        shade +=
          (hash(Math.floor(u * 13 + (row % 2) * 0.5), row, 41) - 0.5) * 0.18;
        shade -= mortar ? 0.27 : 0;
        h += mortar
          ? -0.08
          : Math.sin(a * Math.PI) * Math.sin(b * Math.PI) * 0.08;
      } else {
        const ripple = Math.sin(
          (v * 9 + Math.sin(u * Math.PI * 4) * 0.22) * Math.PI * 2,
        );
        shade += ripple * 0.09 + broad * 0.045;
        h += ripple * 0.13;
      }
      const i = (y * n + x) * 4;
      bytes.set(
        [rgb[0] * shade, rgb[1] * shade, rgb[2] * shade, 255].map((c) =>
          Math.max(0, Math.min(255, Math.round(c))),
        ),
        i,
      );
      heights[y * n + x] = h;
    }
  const at = (x: number, y: number) =>
    heights[((y + n) % n) * n + ((x + n) % n)];
  for (let y = 0; y < n; y++)
    for (let x = 0; x < n; x++) {
      const dx = (at(x - 1, y) - at(x + 1, y)) * 2,
        dy = (at(x, y - 1) - at(x, y + 1)) * 2,
        length = Math.hypot(dx, dy, 1);
      normal.set(
        [
          Math.round(((dx / length) * 0.5 + 0.5) * 255),
          Math.round(((dy / length) * 0.5 + 0.5) * 255),
          Math.round(((1 / length) * 0.5 + 0.5) * 255),
          255,
        ],
        (y * n + x) * 4,
      );
    }
  const tex = (data: Uint8Array, colour: boolean) => {
    const t = new THREE.DataTexture(data, n, n, THREE.RGBAFormat);
    t.wrapS = t.wrapT = THREE.RepeatWrapping;
    t.magFilter = THREE.LinearFilter;
    t.minFilter = THREE.LinearMipmapLinearFilter;
    t.generateMipmaps = true;
    t.anisotropy = 4;
    if (colour) t.colorSpace = THREE.SRGBColorSpace;
    t.needsUpdate = true;
    owned.push(t);
    return t;
  };
  const mat = new THREE.MeshStandardMaterial({
    map: tex(bytes, true),
    normalMap: tex(normal, false),
    roughness: place === 'harbour' ? 0.31 : 0.94,
    metalness: place === 'harbour' ? 0.28 : 0,
    normalScale: new THREE.Vector2(
      place === 'harbour' ? 0.48 : 0.22,
      place === 'harbour' ? 0.48 : 0.22,
    ),
  });
  owned.push(mat);
  const segments = place === 'harbour' ? (quality === 'high' ? 40 : 24) : 1,
    geo = new THREE.PlaneGeometry(180, 180, segments, segments);
  geo.rotateX(-Math.PI / 2);
  const uv = geo.getAttribute('uv');
  for (let i = 0; i < uv.count; i++)
    uv.setXY(i, (uv.getX(i) * 180) / 8, (uv.getY(i) * 180) / 8);
  const mesh = new THREE.Mesh(geo, mat);
  mesh.name = `${place}-authored-ground`;
  mesh.position.set(5.5, place === 'harbour' ? -0.19 : -0.24, -18);
  mesh.receiveShadow = true;
  group.add(mesh);
  if (place === 'harbour') {
    mesh.userData.dynamic = true;
    const positions = geo.getAttribute('position');
    motions.push((time) => {
      for (let i = 0; i < positions.count; i++)
        positions.setY(
          i,
          Math.sin(positions.getX(i) * 0.43 + time * 0.55) * 0.018 +
            Math.sin(positions.getZ(i) * 0.33 - time * 0.4) * 0.012,
        );
      positions.needsUpdate = true;
      geo.computeVertexNormals();
    });
  }
}

/** One instanced draw for all soft lamp/sign pools, no new scene lights. */
function lightPools(
  group: THREE.Group,
  pools: Pool[],
  owned: { dispose(): void }[],
  motions: Motion[],
) {
  if (!pools.length) return;
  const n = 64,
    data = new Uint8Array(n * n * 4);
  for (let y = 0; y < n; y++)
    for (let x = 0; x < n; x++) {
      const distance = Math.hypot(
          ((x + 0.5) / n) * 2 - 1,
          ((y + 0.5) / n) * 2 - 1,
        ),
        alpha = Math.pow(Math.max(0, 1 - distance), 2.3);
      data.set([255, 255, 255, Math.round(alpha * 255)], (y * n + x) * 4);
    }
  const texture = new THREE.DataTexture(data, n, n, THREE.RGBAFormat);
  texture.needsUpdate = true;
  texture.magFilter = texture.minFilter = THREE.LinearFilter;
  const mat = new THREE.MeshBasicMaterial({
    map: texture,
    transparent: true,
    opacity: 0.2,
    blending: THREE.AdditiveBlending,
    depthWrite: false,
    toneMapped: false,
    polygonOffset: true,
    polygonOffsetFactor: -2,
  });
  owned.push(texture, mat);
  const geo = new THREE.PlaneGeometry(1, 1);
  geo.rotateX(-Math.PI / 2);
  const mesh = new THREE.InstancedMesh(geo, mat, pools.length),
    matrix = new THREE.Matrix4(),
    quaternion = new THREE.Quaternion();
  mesh.name = 'soft-night-light-pools';
  mesh.renderOrder = 2;
  for (let i = 0; i < pools.length; i++) {
    const p = pools[i];
    matrix.compose(
      new THREE.Vector3(p.x, p.y ?? -0.221, p.z),
      quaternion,
      new THREE.Vector3(p.w, 1, p.d),
    );
    mesh.setMatrixAt(i, matrix);
    mesh.setColorAt(i, new THREE.Color(p.colour));
  }
  group.add(mesh);
  motions.push((_time, cues) => mat.color.setScalar(lightLevel(cues) * 2.6));
}

function patch(
  k: StageKit,
  at: V3,
  width: number,
  depth: number,
  colour: string,
  wet = false,
) {
  const s = new THREE.Shape();
  for (let i = 0; i < 14; i++) {
    const a = (i / 14) * Math.PI * 2,
      r = 0.81 + Math.sin(i * 2.17 + at[0]) * 0.16,
      x = Math.sin(a) * width * 0.5 * r,
      y = Math.cos(a) * depth * 0.5 * r;
    if (!i) s.moveTo(x, y);
    else s.lineTo(x, y);
  }
  s.closePath();
  const g = new THREE.ShapeGeometry(s);
  g.rotateX(-Math.PI / 2);
  k.add(wet ? 'gloss' : 'paint', colour, g, at);
}

function kerb(k: StageKit, a: V3, b: V3, colour = '#747079') {
  const length = Math.hypot(b[0] - a[0], b[2] - a[2]),
    count = Math.ceil(length / 0.8);
  for (let i = 0; i < count; i++) {
    const t = (i + 0.5) / count;
    k.box(
      'paint',
      i % 6 ? colour : '#9f9788',
      [0.2, 0.16, length / count - 0.025],
      [a[0] + (b[0] - a[0]) * t, -0.18, a[2] + (b[2] - a[2]) * t],
      { rot: [0, Math.atan2(b[0] - a[0], b[2] - a[2]), 0] },
    );
  }
}

/** Small distant people still have shaped shoulders, chin, clothing and shoes. */
function townsperson(quality: StageQuality, sailor = false) {
  const kits = Array.from({ length: 5 }, () => new StageKit(quality));
  const rings = (
    k: StageKit,
    tint: string,
    rows: [number, number, number][],
    at: V3,
  ) => {
    const count = quality === 'high' ? 10 : 7,
      positions: number[] = [],
      uvs: number[] = [],
      index: number[] = [];
    for (const [y, w, d] of rows)
      for (let i = 0; i <= count; i++) {
        const a = (i / count) * Math.PI * 2;
        positions.push(Math.sin(a) * w, y, Math.cos(a) * d);
        uvs.push(i / count, y);
      }
    for (let row = 0; row < rows.length - 1; row++)
      for (let i = 0; i < count; i++) {
        const a = row * (count + 1) + i,
          b = a + count + 1;
        index.push(a, a + 1, b, a + 1, b + 1, b);
      }
    const g = new THREE.BufferGeometry();
    g.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
    g.setIndex(index);
    g.computeVertexNormals();
    k.add('paint', tint, g, at);
  };
  const body = kits[0];
  rings(
    body,
    '#638b94',
    [
      [0.55, 0.12, 0.09],
      [0.7, 0.13, 0.085],
      [0.88, 0.19, 0.09],
      [0.97, 0.14, 0.08],
      [1.0, 0.055, 0.055],
    ],
    [0, 0, 0],
  );
  rings(
    body,
    '#c9a27f',
    [
      [1.01, 0.048, 0.05],
      [1.055, 0.08, 0.07],
      [1.17, 0.1, 0.08],
      [1.24, 0.082, 0.077],
      [1.27, 0.022, 0.025],
    ],
    [0, 0, 0.01],
  );
  rings(
    body,
    '#40373c',
    [
      [1.19, 0.102, 0.077],
      [1.255, 0.087, 0.082],
      [1.28, 0.018, 0.026],
    ],
    [0, 0, -0.005],
  );
  body.box('paint', '#c3b292', [0.105, 0.033, 0.14], [0, 0.947, 0.015]);
  body.box('paint', '#564541', [0.24, 0.025, 0.18], [0, 0.57, 0]);
  if (sailor) {
    body.cyl('paint', '#b3b6a8', 0.09, 0.108, 0.065, [0, 1.264, 0], { seg: 8 });
    body.box('paint', '#384754', [0.21, 0.035, 0.16], [0, 1.22, 0.015]);
  }
  body.cyl('paint', '#d2aa84', 0, 0.027, 0.045, [0, 1.13, 0.103], {
    axis: 'z',
    seg: 6,
  });
  for (const side of [-1, 1]) {
    const leg = kits[side < 0 ? 1 : 2],
      arm = kits[side < 0 ? 3 : 4];
    rings(
      leg,
      '#434758',
      [
        [0.075, 0.045, 0.048],
        [0.29, 0.05, 0.058],
        [0.42, 0.055, 0.063],
        [0.58, 0.071, 0.081],
      ],
      [side * 0.082, 0, 0],
    );
    leg.box(
      'paint',
      '#292e39',
      [0.095, 0.065, 0.18],
      [side * 0.082, 0.04, 0.04],
    );
    rings(
      arm,
      '#638b94',
      [
        [0.63, 0.043, 0.042],
        [0.76, 0.051, 0.049],
        [0.86, 0.062, 0.054],
        [0.96, 0.07, 0.062],
      ],
      [side * 0.175, 0, 0],
    );
    rings(
      arm,
      '#c9a27f',
      [
        [0.58, 0.032, 0.033],
        [0.64, 0.036, 0.04],
      ],
      [side * 0.175, 0, 0],
    );
  }
  const geometries = kits.map((k) => k.build().paint!),
    ranges: [number, number][] = [];
  let offset = 0;
  for (const geo of geometries) {
    ranges.push([offset, geo.getAttribute('position').count]);
    offset += geo.getAttribute('position').count;
  }
  const geo = mergeGeometries(geometries, false)!;
  for (const g of geometries) g.dispose();
  const positions = geo.getAttribute('position'),
    normals = geo.getAttribute('normal'),
    base = Float32Array.from(positions.array),
    baseNormal = Float32Array.from(normals.array);
  const animate = (time: number) => {
    const stride = Math.sin(time * 3.6) * 0.27;
    for (let limb = 1; limb < ranges.length; limb++) {
      const [start, count] = ranges[limb],
        angle = (limb % 2 ? 1 : -1) * stride * (limb > 2 ? -0.6 : 1),
        c = Math.cos(angle),
        s = Math.sin(angle),
        pivot = limb > 2 ? 0.93 : 0.57;
      for (let i = start; i < start + count; i++) {
        const y = base[i * 3 + 1] - pivot,
          z = base[i * 3 + 2];
        positions.setXYZ(i, base[i * 3], y * c - z * s + pivot, y * s + z * c);
        normals.setXYZ(
          i,
          baseNormal[i * 3],
          baseNormal[i * 3 + 1] * c - baseNormal[i * 3 + 2] * s,
          baseNormal[i * 3 + 1] * s + baseNormal[i * 3 + 2] * c,
        );
      }
    }
    positions.needsUpdate = normals.needsUpdate = true;
  };
  return { geo, animate };
}

function crowd(
  group: THREE.Group,
  mats: Materials,
  quality: StageQuality,
  paths: WalkingPath[],
  motions: Motion[],
  sailor = false,
) {
  if (!paths.length) return;
  const figure = townsperson(quality, sailor),
    mesh = new THREE.InstancedMesh(figure.geo, mats.paint, paths.length),
    matrix = new THREE.Matrix4(),
    quaternion = new THREE.Quaternion();
  mesh.name = 'strolling-background-neighbours';
  mesh.userData.dynamic = true;
  // Distant silhouettes use ground pools rather than spending deck shadow draws.
  mesh.castShadow = false;
  mesh.receiveShadow = true;
  const update = (time: number) => {
    figure.animate(time);
    for (let i = 0; i < paths.length; i++) {
      const [x, z, dx, dz, groundY = -0.223] = paths[i],
        phase = time * 0.11 + i * 2.31,
        px = x + Math.sin(phase) * dx,
        pz = z + Math.cos(phase) * dz,
        yaw = Math.atan2(Math.cos(phase) * dx, -Math.sin(phase) * dz),
        scale = i % 6 === 1 ? 0.7 : 0.9 + (i % 3) * 0.06;
      quaternion.setFromAxisAngle(new THREE.Vector3(0, 1, 0), yaw);
      matrix.compose(
        new THREE.Vector3(px, groundY + Math.sin(time * 3.6) ** 2 * 0.012, pz),
        quaternion,
        new THREE.Vector3(scale, scale, scale),
      );
      mesh.setMatrixAt(i, matrix);
      mesh.setColorAt(
        i,
        new THREE.Color(['#d5c2b0', '#afc4bd', '#b8b1ce', '#c3b8a1'][i % 4]),
      );
    }
    mesh.instanceMatrix.needsUpdate = true;
    mesh.computeBoundingSphere();
  };
  update(0);
  group.add(mesh);
  motions.push(update);
}

function distantTraffic(
  group: THREE.Group,
  mats: Materials,
  quality: StageQuality,
  lanes: [number, number, number, number][],
  motions: Motion[],
  route?: THREE.CatmullRomCurve3,
) {
  const car = new StageKit(quality);
  sedan(car, [0, 0, 0], '#8d797f', 0, 1);
  const parts = car.build(),
    solidParts: THREE.BufferGeometry[] = [];
  for (const [name, geo] of Object.entries(parts))
    if (name !== 'glow') solidParts.push(geo);
  const bodyGeo = mergeGeometries(solidParts, false)!;
  for (const g of solidParts) g.dispose();
  const body = new THREE.InstancedMesh(bodyGeo, mats.gloss, lanes.length),
    lamps = new THREE.InstancedMesh(parts.glow!, mats.glow, lanes.length),
    matrix = new THREE.Matrix4(),
    q = new THREE.Quaternion();
  body.name = 'moving-night-highway-cars';
  lamps.name = 'moving-highway-headlights';
  body.castShadow = false;
  group.add(body, lamps);
  const update = (time: number) => {
    for (let i = 0; i < lanes.length; i++) {
      const [z, direction, phase, range] = lanes[i],
        speed = route ? 2.8 : 2.8 + (i % 3) * 0.35,
        x = -range / 2 + ((time * speed + phase) % range);
      const fraction = (x + range / 2) / range,
        point = route?.getPoint(direction < 0 ? 1 - fraction : fraction),
        tangent = route
          ?.getTangent(direction < 0 ? 1 - fraction : fraction)
          .multiplyScalar(direction < 0 ? -1 : 1);
      q.setFromAxisAngle(
        new THREE.Vector3(0, 1, 0),
        tangent
          ? Math.atan2(tangent.z, -tangent.x)
          : direction < 0
            ? 0
            : Math.PI,
      );
      matrix.compose(
        point && tangent
          ? new THREE.Vector3(
              point.x + tangent.z * 0.85,
              -0.218,
              point.z - tangent.x * 0.85,
            )
          : new THREE.Vector3(direction < 0 ? -x : x, -0.225, z),
        q,
        new THREE.Vector3(1.05, 1.05, 1.05),
      );
      body.setMatrixAt(i, matrix);
      lamps.setMatrixAt(i, matrix);
      body.setColorAt(
        i,
        new THREE.Color(['#c9b28d', '#b0b8bd', '#c88e92', '#8bbbad'][i % 4]),
      );
    }
    for (const mesh of [body, lamps]) {
      mesh.instanceMatrix.needsUpdate = true;
      mesh.computeBoundingSphere();
    }
  };
  update(0);
  motions.push(update);
}

function roadRibbon(k: StageKit, points: V3[], width = 4.3) {
  const curve = new THREE.CatmullRomCurve3(
      points.map((p) => new THREE.Vector3(...p)),
      false,
      'centripetal',
    ),
    positions: number[] = [],
    uv: number[] = [],
    indices: number[] = [],
    length = curve.getLength();
  for (let i = 0; i <= 80; i++) {
    const p = curve.getPoint(i / 80),
      tangent = curve.getTangent(i / 80),
      normal = new THREE.Vector3(-tangent.z, 0, tangent.x).normalize();
    for (const side of [-1, 1]) {
      positions.push(
        p.x + ((normal.x * width) / 2) * side,
        -0.214,
        p.z + ((normal.z * width) / 2) * side,
      );
      uv.push(side < 0 ? 0 : 1, (i / 80) * length);
    }
    if (i < 80) {
      const a = i * 2;
      indices.push(a, a + 1, a + 2, a + 1, a + 3, a + 2);
    }
    if (i % 3 === 0) {
      k.box('paint', '#928876', [0.065, 0.006, 1.45], [p.x, -0.203, p.z], {
        rot: [0, Math.atan2(tangent.x, tangent.z), 0],
      });
      k.box(
        'paint',
        '#6d6079',
        [0.065, 0.54, 0.055],
        [
          p.x + normal.x * (width / 2 + 0.38),
          0.028,
          p.z + normal.z * (width / 2 + 0.38),
        ],
      );
    }
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geo.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
  geo.setIndex(indices);
  geo.computeVertexNormals();

  return curve;
}

function steam(
  group: THREE.Group,
  owned: { dispose(): void }[],
  motions: Motion[],
  at: V3,
) {
  const n = 48,
    data = new Uint8Array(n * n * 4);
  for (let y = 0; y < n; y++)
    for (let x = 0; x < n; x++) {
      const d = Math.hypot(((x + 0.5) / n) * 2 - 1, ((y + 0.5) / n) * 2 - 1);
      data.set(
        [255, 255, 255, Math.round(Math.max(0, 1 - d) ** 2 * 255)],
        (y * n + x) * 4,
      );
    }
  const texture = new THREE.DataTexture(data, n, n, THREE.RGBAFormat);
  texture.needsUpdate = true;
  texture.magFilter = texture.minFilter = THREE.LinearFilter;
  const mat = new THREE.MeshBasicMaterial({
    map: texture,
    color: '#9398ae',
    transparent: true,
    opacity: 0.17,
    depthWrite: false,
    side: THREE.DoubleSide,
  });
  owned.push(texture, mat);
  const mesh = new THREE.InstancedMesh(new THREE.PlaneGeometry(1, 1), mat, 5),
    matrix = new THREE.Matrix4(),
    q = new THREE.Quaternion().setFromEuler(new THREE.Euler(-0.48, 0, 0));
  mesh.name = 'service-vent-steam';
  group.add(mesh);
  const update = (time: number) => {
    for (let i = 0; i < 5; i++) {
      const t = (time * 0.22 + i / 5) % 1,
        s = 0.25 + t * 0.7;
      matrix.compose(
        new THREE.Vector3(
          at[0] + t * 0.35 + Math.sin(t * 5) * 0.055,
          at[1] + t * 1.4,
          at[2],
        ),
        q,
        new THREE.Vector3(s, s, s),
      );
      mesh.setMatrixAt(i, matrix);
    }
    mesh.instanceMatrix.needsUpdate = true;
    mesh.computeBoundingSphere();
  };
  update(0);
  motions.push(update);
}

function flutterFlags(
  group: THREE.Group,
  mats: Materials,
  owned: { dispose(): void }[],
  motions: Motion[],
) {
  const geo = new THREE.PlaneGeometry(0.7, 0.35, 8, 3);
  geo.translate(0.35, -0.17, 0);
  const positions = geo.getAttribute('position'),
    base = Float32Array.from(positions.array),
    mat = mats.cloth.clone();
  mat.side = THREE.DoubleSide;
  owned.push(mat);
  const mesh = new THREE.InstancedMesh(geo, mat, 6),
    matrix = new THREE.Matrix4(),
    q = new THREE.Quaternion();
  mesh.name = 'fluttering-midway-flags';
  for (let i = 0; i < 6; i++) {
    matrix.compose(
      new THREE.Vector3(-6 + i * 4.0, 2.55, -11.1),
      q,
      new THREE.Vector3(1, 1, 1),
    );
    mesh.setMatrixAt(i, matrix);
    mesh.setColorAt(
      i,
      new THREE.Color(['#aa567b', '#9e873d', '#568d8c'][i % 3]),
    );
  }
  group.add(mesh);
  motions.push((time) => {
    for (let i = 0; i < positions.count; i++)
      positions.setZ(
        i,
        Math.sin(time * 1.9 - base[i * 3] * 9 + base[i * 3 + 1] * 3) *
          base[i * 3] *
          0.17,
      );
    positions.needsUpdate = true;
    geo.computeVertexNormals();
  });
}

function moonGlitter(
  group: THREE.Group,
  owned: { dispose(): void }[],
  motions: Motion[],
) {

    color: '#6896a5',

      const z = i < 96 ? -7.8 - ((i * 0.61) % 30) : 18 + ((i * 0.87) % 20),
        x = 5.5 + (hash(i, 17) - 0.5) * (5.0 + Math.abs(z) * 0.13),

          1,
          0.018 + hash(i, 8) * 0.028,

    mesh.instanceMatrix.needsUpdate = true;
    mesh.computeBoundingSphere();

    for (const x of [-3.8, 14.5]) {
      rocks(k, x, 14.7, 1.0);
      desertPlants(k, x - 0.6, 15.3, 1.15);
      patch(k, [x, -0.219, 15.1], 5, 3.5, '#695069');
    }
    people.push(
      [2.8, -14.1, 1.0, 0.3],
      [15.6, 8.2, 0.7, 1.0],
      [18.6, 8.1, 0.4, 0.8],
    );
  }
  crowd(group, mats, quality, people, motions, place === 'harbour');
  lightPools(group, pools, owned, motions);
}

export function buildPlace(
  materials: Materials,
  place: PlaceId,
  options: PlaceOptions,
): PlaceModel {
  const stage = options.detail === 'stage',
    quality = options.quality ?? (stage ? 'high' : 'low');
  const group = new THREE.Group();
  group.name = `place-${place}-${options.detail}`;
  const own = glowMaterials(materials),
    mats = own.materials,
    k = new StageKit(quality);
  own.glow.color.setScalar(3);
  const motions: Motion[] = [],
    owned: { dispose(): void }[] = [own.glow],
    landmarks: THREE.Vector3[] = [];
  // Map vignettes use their own compact composition; no ground tile or diner copy.
  const dynamic = (name: string, source: StageKit, position: V3) => {
    const g = joinKit(group, source, mats, name);
    g.position.set(...position);
    g.userData.dynamic = true;
    return g;
  };
  if (stage) {
    groundSurface(group, place, quality, owned, motions);
  }

  if (place === 'truckstop') {
    const h = stage ? 3.65 : 4.5,
      sign: V3 = stage ? [3.3, 0, -3.25] : [0, 0, 0];
    k.cyl('chrome', '#9facb2', 0.13, 0.2, h, [sign[0], h / 2, sign[2]]);
    k.add(
      'gloss',
      '#a9623f',
      panel(stage ? 3.8 : 2.6, stage ? 1.12 : 1, 0.38, 0.17, quality),
      [sign[0], h, sign[2]],
    );
    neon(
      k,
      'FUEL',
      stage ? 0.65 : 0.46,
      [sign[0], h - (stage ? 0.29 : 0.2), sign[2] + 0.12],
      '#ffd084',
    );
    k.rbox('glow', '#f4ae65', [stage ? 4.1 : 2.85, 0.06, 0.08], 0.025, [
      sign[0],
      h - 0.66,
      sign[2] + 0.07,
    ]);
    const px = stage ? -4.9 : -1.4,
      pz = stage ? -3.6 : 1.0,
      s = stage ? 1 : 0.6;
    const pk = new StageKit(quality);
    for (const x of [-1.2, 1.2]) pump(pk, x, 0);
    for (const x of [-2.1, 2.1])
      pk.cyl('chrome', '#b3c4c6', 0.07, 0.07, 2.5, [x, 1.25, 0]);
    pk.rbox('gloss', '#dfbf91', [5.0, 0.25, 2.45], 0.1, [0, 2.5, 0]);
    pk.box('glow', '#ffc588', [4.8, 0.055, 0.06], [0, 2.48, 1.24]);
    const p = joinKit(group, pk, mats, 'pumps');
    p.position.set(px, 0, pz);
    p.scale.setScalar(s);
    if (stage) {
      for (let n = 0; n < 3; n++) {
        const x = 13.9 + n * 5.7,
          z = -5.9 - (n % 2) * 3.1;
        k.rbox(
          'paint',
          ['#586976', '#7e7774', '#5c4358'][n],
          [3.4, 2.7, 7.6],
          0.17,
          [x, 1.43, z],
        );
        k.box('gloss', '#a45742', [3.4, 0.26, 0.03], [x, 1.5, z + 3.82]);
        for (let i = 0; i < 15; i++)
          k.box(
            'chrome',
            '#6c7b8a',
            [0.018, 2.2, 0.018],
            [x - 1.5 + i * 0.215, 1.48, z + 3.815],
          );
        for (const dx of [-1.05, 1.05])
          k.cyl('paint', '#273342', 0.4, 0.4, 0.24, [x + dx, 0.16, z + 2.8], {
            axis: 'x',
            seg: 12,
          });
      }
      lamp(k, -11, -5, 6.4, '#f7b278');
      lamp(k, 23, -5, 6.4, '#f7b278');
    }
    landmarks.push(new THREE.Vector3(sign[0], h + 0.8, sign[2]));
  } else if (place === 'drivein') {
    const center = stage ? ([5.5, 3.0, -3.65] as V3) : ([0, 2.0, 0] as V3);
    const w = stage ? 10.0 : 4.5,
      h = stage ? 3.0 : 2.5;
    k.rbox('paint', '#344964', [w + 0.48, h + 0.5, 0.48], 0.16, center);
    k.rbox('chrome', '#a5c2d2', [w + 0.13, h + 0.13, 0.07], 0.03, [
      center[0],
      center[1],
      center[2] + 0.27,
    ]);
    for (const s of [-1, 1])
      strut(
        k,
        'paint',
        '#627582',
        [center[0] + s * w * 0.42, 0, center[2]],
        [center[0] + s * w * 0.42, center[1] + h * 0.4, center[2]],
        0.14,
      );
    const screenMat = new THREE.MeshBasicMaterial({
      color: '#82bdcc',
      toneMapped: false,
    });
    owned.push(screenMat);
    const screenColour = screenMat.color.clone();
    const screenGeo = panel(w - 0.18, h - 0.16, 0.08, 0.02, quality);
    const screen = new THREE.Mesh(screenGeo, screenMat);
    screen.position.set(center[0], center[1], center[2] + 0.32);
    group.add(screen);
    // Film is a considered physical silhouette of a rocket above layered mountains.
    const film = new StageKit(quality),
      f = stage ? 0.66 : 0.43;
    film.sphere(
      'glow',
      '#ffdfa1',
      0.62 * f,
      [center[0] - 2.9 * f, center[1] + 1.55 * f, center[2] + 0.35],
      { scale: [1, 1, 0.035], seg: 24 },
    );
    for (let n = 0; n < 4; n++) {
      const s = new THREE.Shape();
      s.moveTo(-1.7 * f, 0);
      s.lineTo(-0.6 * f, (0.5 + (n % 2)) * f);
      s.lineTo(0.3 * f, 0.7 * f);
      s.lineTo(1.7 * f, 0);
      s.closePath();
      film.add(
        'glow',
        n % 2 ? '#3e6b8d' : '#527fa0',
        new THREE.ShapeGeometry(s),
        [
          center[0] + (n - 1.5) * 2.3 * f,
          center[1] - h * 0.44,
          center[2] + 0.355,
        ],
      );
    }
    film.capsule(
      'glow',
      '#e9ebd3',
      0.2 * f,
      1.15 * f,
      [center[0] + 1.5 * f, center[1] + 0.3 * f, center[2] + 0.38],
      { rot: [0, 0, -0.4], scale: [1, 1, 0.14] },
    );
    film.tube(
      'glow',
      '#eaaa75',
      [
        [center[0] + 1.22 * f, center[1] - 0.42 * f, center[2] + 0.39],
        [center[0] + 0.58 * f, center[1] - 1.15 * f, center[2] + 0.39],
      ],
      0.1 * f,
    );
    joinKit(group, film, mats, 'movie');
    motions.push((t, cues) => {
      screenMat.color
        .copy(screenColour)
        .multiplyScalar(lightLevel(cues) * (2.5 + Math.sin(t * 11) * 0.04));
    });
    if (stage) {
      for (let row = 0; row < 2; row++)
        for (let i = 0; i < 7; i++) {
          const px = -7.5 + i * 4.5;
          // Rows flank the trailer; a parked car never occupies its closed back.
          if (px > -2 && px < 13) continue;
          sedan(
            k,
            [px, -0.24, -2.8 - row * 2.8],
            ['#b36375', '#a9b9a3', '#719bad', '#bba17d'][i % 4],
            Math.PI / 2,
            1.13,
          );
        }
      k.rbox('paint', '#526479', [2.2, 1.8, 1.8], 0.15, [-7.2, 0.9, 0.8]);
      k.cyl('chrome', '#91b2c6', 0.19, 0.28, 0.5, [-7.2, 1.92, 0.75], {
        axis: 'z',
      });
      const beam = new THREE.Mesh(
        new THREE.ConeGeometry(1.8, 9, 16, 1, true),
        new THREE.MeshBasicMaterial({
          color: '#99c4e9',
          transparent: true,
          opacity: 0.055,
          depthWrite: false,
          side: THREE.DoubleSide,
          blending: THREE.AdditiveBlending,
        }),
      );
      owned.push(beam.material);
      beam.position.set(-4.3, 3.0, -3.4);
      beam.rotation.z = -0.62;
      beam.rotation.x = -0.9;
      group.add(beam);
      motions.push((t, cues) => {
        beam.material.opacity = cues.power
          ? 0.048 + Math.sin(t * 13) * 0.003
          : 0.004;
      });
    } else {
      sedan(k, [-1.4, 0, 1.7], '#b57377', 0.2, 0.7);
      sedan(k, [1.1, 0, 1.9], '#77a59a', -0.1, 0.7);
    }
    landmarks.push(new THREE.Vector3(center[0], center[1] + h / 2, center[2]));
  } else if (place === 'fairground') {
    const x = stage ? 3.3 : 0,
      z = stage ? -4.1 : 0,
      r = 1.65,
      hub = stage ? 2.3 : 2.15;
    for (const s of [-1, 1])
      for (const dz of [-0.4, 0.4])
        strut(
          k,
          'paint',
          '#cf7b9b',
          [x + s * r * 0.8, 0, z + dz * 2],
          [x, hub, z + dz],
          stage ? 0.11 : 0.065,
        );
    const wk = new StageKit(quality);
    wk.torus('gloss', '#cf83a2', r, r * 0.035, [0, 0, 0], {
      rot: [Math.PI / 2, 0, 0],
    });
    wk.torus('glow', '#ffd89b', r * 0.95, r * 0.009, [0, 0, 0.07], {
      rot: [Math.PI / 2, 0, 0],
    });
    const count = stage ? 12 : 8;
    for (let i = 0; i < count; i++) {
      const a = (i / count) * Math.PI * 2;
      strut(
        wk,
        'chrome',
        '#839eb8',
        [0, 0, 0],
        [Math.sin(a) * r, Math.cos(a) * r, 0],
        r * 0.015,
      );
    }
    wk.cyl('chrome', '#c2ccd1', r * 0.13, r * 0.13, 0.35, [0, 0, 0], {
      axis: 'z',
    });
    const wheel = dynamic('ferris-wheel', wk, [x, hub, z]);
    // One batch of gondolas is rebuilt as instances, keeping each cabin upright.
    const cabin = new StageKit(quality),
      s = stage ? 0.72 : 0.53;
    cabin.rbox('gloss', '#79bab2', [0.6 * s, 0.3 * s, 0.46 * s], 0.09 * s, [
      0,
      -0.08 * s,
      0,
    ]);
    cabin.rbox('gloss', '#f3d6a6', [0.68 * s, 0.1 * s, 0.53 * s], 0.05 * s, [
      0,
      0.41 * s,
      0,
    ]);
    for (const dx of [-0.23, 0.23])
      strut(
        cabin,
        'chrome',
        '#bdc8cd',
        [dx * s, 0, 0],
        [dx * s, 0.39 * s, 0],
        0.017 * s,
      );
    const cabins: THREE.InstancedMesh[] = [];
    for (const [mat, geo] of Object.entries(cabin.build())) {
      const mesh = new THREE.InstancedMesh(
        geo,
        mats[mat as keyof Materials],
        count,
      );
      mesh.castShadow = mat !== 'glow';
      group.add(mesh);
      cabins.push(mesh);
    }
    const matrix = new THREE.Matrix4();
    const animate = (t: number) => {
      const angle = t * 0.12;
      wheel.rotation.z = angle;
      for (let i = 0; i < count; i++) {
        const a = (i / count) * Math.PI * 2 - angle;
        matrix.makeTranslation(
          x + Math.sin(a) * r,
          hub + Math.cos(a) * r - 0.15 * s,
          z + 0.25,
        );
        for (const mesh of cabins) mesh.setMatrixAt(i, matrix);
      }
      for (const mesh of cabins) {
        mesh.instanceMatrix.needsUpdate = true;
        mesh.computeBoundingSphere();
      }
    };
    animate(0);
    motions.push(animate);
    const ck = new StageKit(quality);
    tent(ck, [0, 0, 0], stage ? 2.4 : 0.95, '#d39169', stage ? 3.0 : 1.4);
    ck.cyl(
      'gloss',
      '#bf7897',
      stage ? 2.3 : 0.9,
      stage ? 2.3 : 0.9,
      0.18,
      [0, 0.11, 0],
      { seg: 24 },
    );
    const carousel = dynamic(
      'carousel',
      ck,
      stage ? [13.9, 0, -4.4] : [2.65, 0, 0.1],
    );
    const horses = new StageKit(quality);
    for (let i = 0; i < 6; i++) {
      const a = (i / 6) * Math.PI * 2,
        rad = stage ? 1.5 : 0.56,
        px = Math.sin(a) * rad,
        pz = Math.cos(a) * rad;
      strut(
        horses,
        'chrome',
        '#e5c383',
        [px, 0.15, pz],
        [px, stage ? 2.3 : 1.0, pz],
        0.025,
      );
      carouselHorse(
        horses,
        [px, stage ? 0.96 : 0.48, pz],
        a + Math.PI / 2,
        stage ? 0.85 : 0.38,
        i % 2 ? '#e1bc90' : '#9dd3c2',
      );
    }
    joinKit(carousel, horses, mats, 'carousel-mounts');
    motions.push((t) => {
      carousel.rotation.y = t * 0.16;
    });
    if (stage) tent(k, [-7.2, 0, -5.2], 3.6, '#728aa9', 4.6);
    landmarks.push(new THREE.Vector3(x, hub + r + 0.4, z));
  } else if (place === 'harbour') {
    const lighthouse = new StageKit(quality),
      h = stage ? 3.5 : 4.2,
      x = stage ? 8.1 : 0,
      z = stage ? -4.3 : 0,
      towerScale = stage ? 0.79 : 1;
    for (let i = 0; i < 5; i++)
      lighthouse.cyl(
        'paint',
        i % 2 ? '#b36368' : '#dcd7c3',
        0.56 + (4 - i) * 0.075,
        0.56 + (5 - i) * 0.075,
        h / 5,
        [0, ((i + 0.5) * h) / 5, 0],
        { seg: 24 },
      );
    lighthouse.cyl('chrome', '#6b909c', 0.96, 0.83, 0.16, [0, h, 0], {
      seg: 24,
    });
    lighthouse.cyl('glow', '#f2d6a0', 0.54, 0.54, 0.76, [0, h + 0.45, 0], {
      seg: 12,
    });
    for (let i = 0; i < 8; i++)
      strut(
        lighthouse,
        'paint',
        '#344c62',
        [
          Math.sin((i * Math.PI) / 4) * 0.58,
          h + 0.1,
          Math.cos((i * Math.PI) / 4) * 0.58,
        ],
        [
          Math.sin((i * Math.PI) / 4) * 0.58,
          h + 0.85,
          Math.cos((i * Math.PI) / 4) * 0.58,
        ],
        0.035,
      );
    lighthouse.cyl('gloss', '#416d79', 0, 1.0, 0.55, [0, h + 1.06, 0], {
      seg: 16,
    });
    lighthouse.add(
      'wood',
      '#564b50',
      panel(0.48, 0.9, 0.18, 0.055, quality),
      [0, 0.52, 0.9],
    );
    const tower = joinKit(group, lighthouse, mats, 'lighthouse');
    tower.position.set(x, 0, z);
    tower.scale.setScalar(towerScale);
    const sweep = new THREE.Group();
    sweep.userData.dynamic = true;
    sweep.position.set(x, (h + 0.45) * towerScale, z);
    group.add(sweep);
    const beamMat = new THREE.MeshBasicMaterial({
      color: '#a6e0d5',
      transparent: true,
      opacity: 0.065,
      depthWrite: false,
      side: THREE.DoubleSide,
      blending: THREE.AdditiveBlending,
    });
    owned.push(beamMat);
    const beam = new THREE.Mesh(
      new THREE.ConeGeometry(stage ? 3 : 1.2, stage ? 17 : 6.5, 16, 1, true),
      beamMat,
    );
    beam.rotation.z = Math.PI / 2;
    beam.position.x = stage ? 8.5 : 3.25;
    sweep.add(beam);
    motions.push((t, cues) => {
      sweep.rotation.y = t * 0.23;
      beamMat.opacity = 0.065 * lightLevel(cues);
    });
    if (stage) {
      // Water is continuous; piers meet the patio apron without a moat.
      for (let i = 0; i < 44; i++)
        k.box(
          'wood',
          i % 3 ? '#8b8179' : '#a1917d',
          [16, 0.07, 0.38],
          [5.5, -0.08, -3.8 + i * 0.4],
        );
      for (const px of [-3.1, 14.1])
        for (const pz of [-2.7, 4.0, 10.7]) {
          k.cyl('wood', '#5a635e', 0.16, 0.2, 1.0, [px, 0, pz]);
          k.torus('chrome', '#96aba7', 0.18, 0.025, [px, 0.42, pz]);
        }
      shop(k, -9.5, -7, 5.2, 2.6, '#5b7e82');
      k.box('wood', '#5c574e', [5.8, 0.08, 3.0], [-9.5, -0.04, -7]);
    }
    for (let n = 0; n < (stage ? 3 : 1); n++) {
      const bk = new StageKit(quality);
      boat(bk, ['#b57063', '#81afa4', '#667996'][n]);
      const b = dynamic(
        'fishing-boat',
        bk,
        stage ? [-5.1 + n * 11.8, -0.18, -3.6 - (n % 2) * 3.5] : [2.1, 0, 0.7],
      );
      if (!stage) b.scale.setScalar(0.65);
      const baseY = b.position.y;
      motions.push((t) => {
        b.position.y = baseY + Math.sin(t * 1.3 + n * 1.8) * 0.07;
        b.rotation.x = Math.sin(t * 1.15 + n) * 0.032;
        b.rotation.z = Math.sin(t * 0.9 + n) * 0.028;
      });
    }
    landmarks.push(new THREE.Vector3(x, (h + 1.4) * towerScale, z));
  } else if (place === 'oldtown') {
    const clock = stage ? ([9.03, 0, -4.1] as V3) : ([0, 0, 0] as V3);
    const h = stage ? 3.7 : 3.1,
      w = stage ? 1.65 : 1.4,
      roofHeight = stage ? 0.8 : 1.6;
    k.rbox('paint', '#917576', [w, h, w], 0.065, [clock[0], h / 2, clock[2]]);
    for (let i = 0; i < 4; i++)
      k.box(
        'ceramic',
        '#b7aa95',
        [w + 0.2, 0.11, w + 0.2],
        [clock[0], h * (0.21 + i * 0.19), clock[2]],
      );
    k.cyl(
      'paint',
      '#384c5f',
      0,
      w * 0.82,
      roofHeight,
      [clock[0], h + roofHeight / 2, clock[2]],
      { seg: 4, rot: [0, Math.PI / 4, 0] },
    );
    const radius = w * 0.35,
      cy = h - radius - 0.12,
      cz = clock[2] + w / 2 + 0.21;
    k.add(
      'paint',
      '#80686b',
      panel(radius * 2 + 0.22, radius * 2 + 0.22, 0.15, 0.24, quality),
      [clock[0], cy, cz - 0.14],
    );
    k.cyl(
      'chrome',
      '#cab992',
      radius + 0.07,
      radius + 0.07,
      0.08,
      [clock[0], cy, cz],
      { axis: 'z', seg: 32 },
    );
    k.cyl('glow', '#e4c894', radius, radius, 0.09, [clock[0], cy, cz], {
      axis: 'z',
      seg: 32,
    });
    for (let i = 0; i < 12; i++)
      k.box(
        'paint',
        '#5b5151',
        [0.02, 0.095, 0.02],
        [
          clock[0] + Math.sin((i * Math.PI) / 6) * radius * 0.79,
          cy + Math.cos((i * Math.PI) / 6) * radius * 0.79,
          cz + 0.06,
        ],
        { rot: [0, 0, (-i * Math.PI) / 6] },
      );
    const hand = new StageKit(quality);
    hand.box(
      'paint',
      '#4a4e58',
      [0.045, radius * 0.66, 0.035],
      [0, radius * 0.23, 0],
    );
    hand.box(
      'paint',
      '#4a4e58',
      [radius * 0.5, 0.046, 0.04],
      [radius * 0.18, 0, 0],
    );
    const hands = dynamic('clock-hands', hand, [clock[0], cy, cz + 0.08]);
    motions.push((t) => {
      hands.rotation.z = -t * 0.007;
    });
    if (stage) {
      for (const [px, height, tint] of [
        [-7.9, 3.8, '#946f66'],
        [-3.2, 4.7, '#78817e'],
        [12.3, 4.2, '#877485'],
        [17.0, 3.4, '#9c826c'],
      ] as [number, number, string][])
        shop(k, px, -7.1, 4.3, height, tint);
      for (const px of [-4.5, 15.5])
        for (const pz of [-3, 11.2]) lamp(k, px, pz, 4.2);
    } else {
      const sk = new StageKit(quality);
      shop(sk, -2.3, 0.2, 2.6, 2.4, '#9a776c');
      joinKit(group, sk, mats, 'shopfront');
      lamp(k, 1.8, 1.2, 2.0);
    }
    landmarks.push(new THREE.Vector3(clock[0], h + roofHeight, clock[2]));
  } else {
    // Sandstone bodies with angled strata and a retained, legible summit.
    if (stage) {
      mesa(k, [-8, -0.23, -16], 8, 5.5, '#6e5872');
      mesa(k, [18, -0.23, -15], 7, 7.1, '#75637f');
      mesa(k, [1, -0.23, -26], 8, 6.3, '#554b69');
    } else mesa(k, [-2.2, 0, -1.5], 1.2, 1.6, '#80687e');
    const x = stage ? 2.7 : 0,
      z = stage ? -5.0 : 0,
      sc = stage ? 0.72 : 0.48;
    k.rbox('paint', '#626c84', [1.6 * sc, 0.24 * sc, 1.5 * sc], 0.1 * sc, [
      x,
      0.1 * sc,
      z,
    ]);
    for (const dx of [-0.62, 0.62])
      strut(
        k,
        'chrome',
        '#98adb9',
        [x + dx * sc, 0.2 * sc, z],
        [x, 3.1 * sc, z],
        0.1 * sc,
      );
    const dish = new THREE.Group();
    dish.userData.dynamic = true;
    dish.position.set(x, 3.25 * sc, z);
    dish.rotation.x = -0.55;
    group.add(dish);
    const dk = new StageKit(quality);
    const bowl = new THREE.LatheGeometry(
      [
        new THREE.Vector2(0.0, -0.78),
        new THREE.Vector2(0.36, -0.74),
        new THREE.Vector2(0.95, -0.59),
        new THREE.Vector2(1.54, -0.3),
        new THREE.Vector2(2.05, 0),
        new THREE.Vector2(2.06, 0.07),
        new THREE.Vector2(1.53, -0.23),
        new THREE.Vector2(0.95, -0.52),
        new THREE.Vector2(0.35, -0.67),
        new THREE.Vector2(0, -0.7),
      ],
      quality === 'high' ? 40 : 20,
    );
    dk.add('chrome', '#b4c3cd', bowl, [0, 0, 0], { scale: [sc, sc, sc] });
    dk.torus('chrome', '#d1d4d0', 2.05 * sc, 0.045 * sc, [0, 0.035 * sc, 0]);
    for (let i = 0; i < 3; i++) {
      const a = (i / 3) * Math.PI * 2;
      strut(
        dk,
        'chrome',
        '#8195a7',
        [Math.sin(a) * 1.93 * sc, 0, Math.cos(a) * 1.93 * sc],
        [0, 1.2 * sc, 0],
        0.032 * sc,
      );
    }
    dk.cyl('gloss', '#866a84', 0.1 * sc, 0.1 * sc, 0.31 * sc, [0, 1.2 * sc, 0]);
    joinKit(dish, dk, mats, 'radio-bowl');
    motions.push((t) => {
      dish.rotation.y = Math.sin(t * 0.1) * 0.12;
    });
    const rocket = new StageKit(quality),
      rx = stage ? 12.4 : 1.8,
      rz = stage ? -4.2 : 0.4,
      rh = stage ? 3.4 : 2.6;
    rocket.cyl('paint', '#514d70', 0.1, 0.15, rh * 0.65, [rx, rh * 0.32, rz]);
    rocket.capsule(
      'gloss',
      '#d4c3b4',
      rh * 0.14,
      rh * 0.4,
      [rx, rh * 0.8, rz],
      { rot: [0, 0, -0.24] },
    );
    for (const dx of [-1, 1]) {
      const s = new THREE.Shape();
      s.moveTo(0, 0);
      s.lineTo(dx * rh * 0.23, -rh * 0.16);
      s.lineTo(dx * rh * 0.19, rh * 0.16);
      s.closePath();
      rocket.add(
        'gloss',
        '#b55c85',
        new THREE.ExtrudeGeometry(s, { depth: 0.09, bevelEnabled: false }),
        [rx, rh * 0.58, rz],
      );
    }
    rocket.sphere(
      'glow',
      '#85d6cb',
      rh * 0.085,
      [rx - rh * 0.03, rh * 0.86, rz + rh * 0.14],
      { scale: [1, 1, 0.2], seg: 16 },
    );
    const rg = joinKit(group, rocket, mats, 'rocket-sign');
    rg.userData.dynamic = true;
    motions.push((t) => {
      rg.rotation.z = Math.sin(t * 0.6) * 0.0015;
    });
    if (stage) {
      const moon = new StageKit(quality);
      // Celestial disc behind the parked car, never a sphere inside its roof.
      moon.sphere('glow', '#c7bdca', 2.05, [0, 0, 0], {
        scale: [1, 1, 0.05],
        seg: 40,
      });
      for (let i = 0; i < 8; i++) {
        const px = Math.sin(i * 2.17) * (0.5 + (i % 3) * 0.25),
          py = Math.cos(i * 2.17) * (0.4 + (i % 3) * 0.25);
        const depth = Math.sqrt(2.05 ** 2 - px ** 2 - py ** 2);
        moon.sphere(
          'glow',
          '#a99fb8',
          0.12 + (i % 3) * 0.075,
          [px, py, depth * 0.05 + 0.012],
          { scale: [1.1, 1, 0.08], seg: 16 },
        );
      }
      const lunarDisc = joinKit(group, moon, materials, 'moon');
      lunarDisc.position.set(8.3, 1.7, -4.5);
      lunarDisc.rotation.x = (-44 * Math.PI) / 180;
      landmarks.push(new THREE.Vector3(8.3, 3.25, -5.93));
      for (const px of [-4, 17, 22]) {
        k.cyl('paint', '#527f78', 0.1, 0.17, 1.45, [px, 0.5, 4.8]);
        k.tube(
          'paint',
          '#527f78',
          [
            [px, 0.45, 4.8],
            [px + 0.38, 0.6, 4.8],
            [px + 0.38, 1.1, 4.8],
          ],
          0.095,
        );
      }
    }
    landmarks.push(new THREE.Vector3(x, 5.1 * sc, z));
  }
  if (stage) {
    const district = new StageKit(quality);
    stageDistrict(place, district, group, mats, quality, owned, motions);
    const scenery = joinKit(
      group,
      district,
      mats,
      'night-surrounding-district',
    );
    // These distant surfaces receive moon/deck light, but only the near hero
    // props spend the single deck shadow pass. Keep this material batch intact.
    scenery.userData.dynamic = true;
    scenery.traverse((o) => {
      if (o instanceof THREE.Mesh) o.castShadow = false;
    });
  } else mapDistrict(k, place);
  joinKit(group, k, mats, 'place-static');
  return {
    group,
    landmarks,
    update(cues) {
      const time = cues.reduced ? 0 : cues.time;
      own.glow.color.setScalar(lightLevel(cues, 1.3) * 3.0);
      for (const motion of motions) motion(time, cues);
    },
    dispose: modelDisposer(group, owned),
  };
}

export function buildMapStop(
  materials: Materials,
  kind: 'garage' | 'motel',
): PlaceModel {
  const group = new THREE.Group();
  group.name = `map-${kind}`;
  const own = glowMaterials(materials),
    k = new StageKit('low');
  own.glow.color.setScalar(3);
  if (kind === 'garage') {
    k.rbox('gloss', '#77a99c', [3.5, 1.7, 2.3], 0.12, [0, 0.85, 0]);
    k.rbox('gloss', '#e0cdab', [3.8, 0.2, 2.5], 0.07, [0, 1.75, 0]);
    k.box('paint', '#2e4151', [1.7, 1.25, 0.06], [-0.35, 0.68, 1.17]);
    for (const dx of [-1.28, 0.56])
      k.box('chrome', '#a0b7bb', [0.07, 1.3, 0.08], [dx, 0.7, 1.22]);
    sedan(k, [-0.35, 0.03, 1.3], '#bd6976', Math.PI / 2, 0.67);
    k.box('gloss', '#bd6976', [0.52, 0.06, 0.56], [-0.35, 0.65, 1.96], {
      rot: [-0.8, 0, 0],
    });
    neon(k, 'GARAGE', 0.38, [0, 2.1, 0], '#b3edd4');
    k.cyl('chrome', '#839da9', 0.05, 0.05, 2.7, [2.2, 1.35, 0.2]);
    k.cyl('gloss', '#9c425c', 0.49, 0.49, 0.12, [2.2, 2.55, 0.2], {
      axis: 'z',
      seg: 32,
    });
    const wrench = new THREE.Shape();
    wrench.moveTo(-0.055, -0.35);
    wrench.lineTo(0.055, -0.35);
    wrench.lineTo(0.08, 0.04);
    wrench.lineTo(0.25, 0.18);
    wrench.lineTo(0.23, 0.36);
    wrench.lineTo(0.13, 0.23);
    wrench.lineTo(-0.01, 0.18);
    wrench.lineTo(-0.12, 0.28);
    wrench.lineTo(-0.15, 0.42);
    wrench.lineTo(-0.28, 0.23);
    wrench.lineTo(-0.19, 0.1);
    wrench.lineTo(-0.07, 0.04);
    wrench.closePath();
    k.add(
      'glow',
      '#f0c191',
      new THREE.ExtrudeGeometry(wrench, { depth: 0.025, bevelEnabled: false }),
      [2.2, 2.53, 0.27],
    );
  } else {
    k.rbox('paint', '#c1b295', [4.7, 1.35, 1.25], 0.04, [0, 0.68, -0.15]);
    k.rbox('paint', '#c1b295', [1.25, 1.35, 2.8], 0.04, [-1.72, 0.68, 0.7]);
    k.box('gloss', '#a75664', [4.9, 0.14, 1.55], [0, 1.4, -0.15]);
    for (let i = 0; i < 5; i++) {
      k.box('paint', '#546776', [0.3, 0.88, 0.05], [-1.6 + i * 0.8, 0.48, 0.5]);
      k.box('glow', '#e3ba86', [0.32, 0.4, 0.06], [-1.2 + i * 0.8, 0.8, 0.5]);
    }
    k.rbox('ceramic', '#dcc8a5', [2.3, 0.13, 1.2], 0.15, [0.6, 0.065, 1.6]);
    k.rbox('gloss', '#5eaaaa', [2.0, 0.05, 0.9], 0.13, [0.6, 0.15, 1.6]);
    k.cyl('chrome', '#9aacb0', 0.05, 0.07, 2.9, [2.75, 1.45, 0]);
    k.add(
      'gloss',
      '#9c425c',
      panel(2.6, 0.72, 0.16, 0.1, 'low'),
      [2.75, 2.9, 0],
    );
    neon(k, 'MOTEL', 0.46, [2.75, 2.66, 0.07], '#ffb38c');
  }
  joinKit(group, k, own.materials, kind);
  return {
    group,
    landmarks: [new THREE.Vector3(0, 3.3, 0)],
    update(cues) {
      own.glow.color.setScalar(3 * lightLevel(cues, 0.4));
    },
    dispose: modelDisposer(group, [own.glow]),
  };
}
