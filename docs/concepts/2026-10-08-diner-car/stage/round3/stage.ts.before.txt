function booth(k: StageKit, t: TableSlot) {
  const length = boothHz(t.seats) * 2 - 0.035;
  k.rbox('chrome', C.chrome, [0.74, 0.056, length - 0.08], 0.025, [
    t.x,
    STAGE.tableTop - 0.047,
    t.z,
  ]);
  k.rbox('laminate', '#ecddbf', [0.7, 0.039, length - 0.12], 0.019, [
    t.x,
    STAGE.tableTop - 0.0195,
    t.z,
  ]);
  k.cyl('chrome', '#c1cfd0', 0.048, 0.06, 0.65, [t.x, 0.349, t.z]);
  k.cyl('chrome', '#b0c0c7', 0.18, 0.22, 0.046, [t.x, 0.023, t.z]);
  for (const side of [-1, 1]) {
    const x = t.x + side * 0.62;
    k.rbox('paint', '#35685f', [0.38, 0.23, length - 0.01], 0.025, [
      x,
      0.23,
      t.z,
    ]);
    k.rbox('chrome', C.chrome, [0.43, 0.035, length], 0.017, [x, 0.353, t.z]);
    const seat = new THREE.Shape();
    seat.moveTo(-0.21, 0.38);
    seat.quadraticCurveTo(-0.225, 0.465, -0.13, 0.468);
    seat.bezierCurveTo(-0.08, 0.47, 0.09, 0.47, 0.16, 0.468);
    seat.quadraticCurveTo(0.23, 0.46, 0.21, 0.385);
    seat.closePath();
    const g = new THREE.ExtrudeGeometry(seat, {
      depth: length - 0.02,
      bevelEnabled: false,
      curveSegments: k.quality === 'high' ? 12 : 6,
    });
    g.translate(0, 0, -(length - 0.02) / 2);
    g.computeBoundingBox();
    g.translate(0, STAGE.seatTop - g.boundingBox!.max.y, 0);
    k.add('vinyl', C.mint, g, [x, 0, t.z]);
    const back = new THREE.Shape();
    back.moveTo(0.12, 0.42);
    back.bezierCurveTo(0.1, 0.6, 0.13, 0.85, 0.145, 0.98);
    back.quadraticCurveTo(0.16, 1.07, 0.22, 1.01);
    back.lineTo(0.225, 0.41);
    back.closePath();
    const bg = new THREE.ExtrudeGeometry(back, {
      depth: length - 0.025,
      bevelEnabled: true,
      bevelThickness: 0.012,
      bevelSize: 0.012,
      bevelSegments: k.quality === 'high' ? 3 : 2,
      curveSegments: k.quality === 'high' ? 12 : 6,
    });
    bg.translate(0, 0, -(length - 0.025) / 2);
    k.add('vinyl', C.mint, bg, [x, 0, t.z], {
      rot: [0, side < 0 ? Math.PI : 0, 0],
    });
    k.tube(
      'chrome',
      '#b0d3cb',
      [
        [x + side * 0.22, 0.45, t.z - length / 2 + 0.015],
        [x + side * 0.22, 1.02, t.z - length / 2 + 0.015],
        [x + side * 0.22, 1.02, t.z + length / 2 - 0.015],
        [x + side * 0.22, 0.45, t.z + length / 2 - 0.015],
      ],
      0.009,
    );
    const channels = t.seats === 2 ? 3 : 5;
    for (let i = 1; i < channels; i++)
      k.tube(
        'vinyl',
        '#65a993',
        [
          [x + side * 0.15, 0.53, t.z - length / 2 + (i / channels) * length],
          [x + side * 0.12, 0.78, t.z - length / 2 + (i / channels) * length],
          [x + side * 0.155, 0.995, t.z - length / 2 + (i / channels) * length],
        ],
        0.004,
      );
    for (const dz of [-1, 1])
      k.box(
        'chrome',
        C.chrome,
        [0.4, 0.025, 0.07],
        [x, 0.03, t.z + dz * length * 0.3],

  }
}

function planter(target: StageKit, x: number, z: number) {
  const k = new StageKit(target.quality);
  k.cyl('ceramic', '#a86d67', 0.26, 0.19, 0.47, [x, 0.22, z], { seg: 16 });
  k.cyl('paint', '#404d43', 0.23, 0.23, 0.018, [x, 0.46, z], { seg: 16 });
  for (let i = 0; i < 7; i++) {
    const a = i * 2.4;
    const stem: [number, number, number][] = [
      [x, 0.44, z],
      [x + Math.sin(a) * 0.09, 0.7, z + Math.cos(a) * 0.09],
      [x + Math.sin(a) * 0.3, 0.92 + (i % 2) * 0.1, z + Math.cos(a) * 0.3],
    ];
    k.tube('paint', '#426a5c', stem, 0.015);
    k.sphere('paint', i % 2 ? '#6b8f68' : '#759778', 0.12, stem[2], {
      scale: [0.6, 1.5, 0.2],
      rot: [0.35, a, 0.35],
      seg: 10,
    });
  }
  for (const [mat, part] of Object.entries(k.build())) {
    part.translate(0, -0.225, 0);
    (target.parts[mat as keyof typeof target.parts] ??= []).push(part);
  }
}

/** Made once per stage, with two tiles per repeat and restrained grout. */
function checker(quality: StageQuality) {
  const n = quality === 'high' ? 256 : 128,
    bytes = new Uint8Array(n * n * 4);
  for (let y = 0; y < n; y++)
    for (let x = 0; x < n; x++) {
      const cell = n / 2,
        colour =
          (Math.floor(x / cell) + Math.floor(y / cell)) % 2
            ? [185, 130, 123]
            : [215, 195, 166];
      const shade = x % cell < 1 || y % cell < 1 ? 0.89 : 1;
      bytes.set(
        [...colour.map((v) => Math.round(v * shade)), 255],
        (y * n + x) * 4,
      );
    }
  const tex = new THREE.DataTexture(bytes, n, n, THREE.RGBAFormat);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
  tex.repeat.set((W + 0.4) / 1.6, (FRONT_Z - COUNTER_Z + 0.4) / 1.6);
  tex.minFilter = THREE.LinearMipmapLinearFilter;
  tex.magFilter = THREE.LinearFilter;
  tex.generateMipmaps = true;
  tex.anisotropy = 4;
  tex.needsUpdate = true;
  return tex;
}

function festoon(
  k: StageKit,
  bulbs: StageKit,
  start: [number, number, number],
  end: [number, number, number],
  sag: number,
  count: number,
) {
  const at = (u: number): [number, number, number] => [
    start[0] + (end[0] - start[0]) * u,
    start[1] +
      (end[1] - start[1]) * u -
      (sag * (Math.cosh(1.3) - Math.cosh(1.3 * (2 * u - 1)))) /
        (Math.cosh(1.3) - 1),
    start[2] + (end[2] - start[2]) * u,
  ];
  k.tube(
    'paint',
    '#1c2b34',
    Array.from({ length: 25 }, (_, i) => at(i / 24)),
    0.019,
  );
  const pear = [
    new THREE.Vector2(0.019, 0),
    new THREE.Vector2(0.036, -0.026),
    new THREE.Vector2(0.073, -0.081),
    new THREE.Vector2(0.076, -0.121),
    new THREE.Vector2(0.052, -0.161),
    new THREE.Vector2(0.008, -0.179),
    new THREE.Vector2(0, -0.178),
  ];
  for (let i = 0; i < count; i++) {
    const [x, y, z] = at((i + 0.5) / count);
    k.cyl('paint', '#283336', 0.021, 0.023, 0.12, [x, y - 0.05, z], { seg: 6 });
    k.cyl('ceramic', '#7e7c68', 0.035, 0.027, 0.042, [x, y - 0.12, z], {
      seg: 8,
    });
    bulbs.add(
      'glow',
      i % 4 === 0 ? '#ffb86a' : '#ffd791',
      new THREE.LatheGeometry(pear, k.quality === 'high' ? 12 : 7),
      [x, y - 0.14, z],
    );
  }
}

function lightPools(group: THREE.Group, floor: number) {
  const n = 128,
    bytes = new Uint8Array(n * n * 4);
  for (let y = 0; y < n; y++)
    for (let x = 0; x < n; x++) {
      const radius = Math.hypot(
        ((x + 0.5) / n) * 2 - 1,
        ((y + 0.5) / n) * 2 - 1,
      );
      const a = Math.max(0, 1 - radius * radius) ** 3;
      bytes.set([255, 255, 255, Math.round(a * 255)], (y * n + x) * 4);
    }
  const texture = new THREE.DataTexture(bytes, n, n, THREE.RGBAFormat);
  texture.needsUpdate = true;
  const material = new THREE.MeshBasicMaterial({
    map: texture,
    color: '#ffc278',
    transparent: true,
    opacity: 0.13,
    blending: THREE.AdditiveBlending,
    depthWrite: false,
    toneMapped: false,
  });
  const geometry = new THREE.PlaneGeometry(1, 1);
  const pools = new THREE.InstancedMesh(geometry, material, 5);
  pools.name = 'patio-lamp-pools';
  const object = new THREE.Object3D();
  for (const [i, p] of [
    [-0.67, 3.1, 2.8, 4.5],
    [W + 0.67, 3.1, 2.8, 4.5],
    [-0.7, FRONT_Z + 0.7, 3, 3.2],
    [W + 0.7, FRONT_Z + 0.7, 3, 3.2],
    [-5.5, -1.4, 3, 2.4],
  ].entries()) {
    object.position.set(p[0], floor, p[1]);
    object.rotation.set(-Math.PI / 2, 0, 0);
    object.scale.set(p[2], p[3], 1);
    object.updateMatrix();
    pools.setMatrixAt(i, object.matrix);
  }
  pools.instanceMatrix.needsUpdate = true;
  group.add(pools);
  return { texture, material };
}

/** Combine only unmoving vertex-coloured geometry; rigs retain their pivots. */
function batch(group: THREE.Group) {
  group.updateMatrixWorld(true);
  const byMat = new Map<THREE.Material, THREE.BufferGeometry[]>(),
    originals: THREE.Mesh[] = [];
  const walk = (node: THREE.Object3D) => {
    if (node.userData.dynamic || node instanceof THREE.InstancedMesh) return;
    if (node instanceof THREE.Mesh && !Array.isArray(node.material)) {
      if (
        Object.keys(node.geometry.attributes).sort().join(',') !==
        'color,normal,position,uv'
      )
        return;
      let geo = node.geometry.clone();
      geo.applyMatrix4(node.matrixWorld);
      if (geo.index) {
        const flat = geo.toNonIndexed();
        geo.dispose();
        geo = flat;
      }
      if (!byMat.has(node.material)) byMat.set(node.material, []);
      byMat.get(node.material)!.push(geo);
      originals.push(node);
    }
    for (const child of node.children) walk(child);
  };
  walk(group);
  const freed = new Set<THREE.BufferGeometry>();
  for (const o of originals) {
    o.removeFromParent();
    if (!freed.has(o.geometry)) {
      freed.add(o.geometry);
      o.geometry.dispose();
    }
  }
  for (const [mat, list] of byMat) {
    const geo = mergeGeometries(list, false);
    for (const g of list) g.dispose();
    if (!geo) continue;
    geo.computeBoundingSphere();
    const mesh = new THREE.Mesh(geo, mat);
    mesh.name = 'stage-batch';
    mesh.castShadow =
      !(mat instanceof THREE.MeshBasicMaterial) && !mat.transparent;
    mesh.receiveShadow = !(mat instanceof THREE.MeshBasicMaterial);
    group.add(mesh);
  }
}

    musicOwn = glowMaterials(materials);

  musicOwn.glow.color.setScalar(2.6);
  const car = buildDinerCar(materials, { detail: 'stage', upgrades, quality });
  // The portable deck stays at y=0; the rig stands on this place's actual surface.
  car.group.position.set(W / 2, options.place === 'harbour' ? 0.195 : 0, 0);

    detail: 'stage',

    k.box(
      'gloss',
      C.cherry,
      [Math.abs(x1 - x0) + 0.03, 0.045, Math.abs(z1 - z0) + 0.03],
      [(x0 + x1) / 2, 0.17, (z0 + z1) / 2],
    );
    for (let n = 0; n <= count; n++)
      k.cyl(
        'chrome',
        '#a4b6b9',
        0.025,
        0.034,
        0.54,
        [x0 + ((x1 - x0) * n) / count, 0.27, z0 + ((z1 - z0) * n) / count],
        { seg: 10 },
      );
  };
  rail(-0.13, COUNTER_Z, -0.13, DOOR_GAP[0]);
  rail(-0.13, DOOR_GAP[1], -0.13, FRONT_Z);
  rail(W + 0.13, COUNTER_Z, W + 0.13, FRONT_Z);
  rail(-0.13, FRONT_Z + 0.1, W + 0.13, FRONT_Z + 0.1);
  const gate = new THREE.Group();
  gate.name = 'entrance-gate';
  gate.userData.dynamic = true;
  gate.position.set(-0.13, 0, DOOR_GAP[0]);
  group.add(gate);
  const gk = new StageKit(quality),
    gap = DOOR_GAP[1] - DOOR_GAP[0];
  gk.rbox('gloss', C.cherry, [0.065, 0.26, gap - 0.06], 0.03, [
    0,
    0.31,
    gap / 2,
  ]);
  gk.box('chrome', '#c3c4ac', [0.07, 0.04, gap - 0.06], [0, 0.48, gap / 2]);
  joinKit(gate, gk, materials, 'gate-panel');
  k.box(
    'chrome',
    '#9baaa6',
    [0.6, 0.26, 0.55],
    [-0.75, -0.11, DOOR_GAP[0] - 0.39],
  );
  k.rbox('gloss', C.mintDark, [0.47, 0.87, 0.44], 0.07, [
    -0.75,
    0.43,
    DOOR_GAP[0] - 0.39,
  ]);
  k.rbox(
    'wood',
    '#dfc3a0',
    [0.59, 0.065, 0.52],
    0.03,
    [-0.75, 0.9, DOOR_GAP[0] - 0.39],
    { rot: [-0.08, 0, 0] },
  );
  k.box(
    'ceramic',
    '#e6dec5',
    [0.28, 0.01, 0.24],
    [-0.75, 0.955, DOOR_GAP[0] - 0.39],
  );
  k.rbox('chrome', '#a8b6b2', [1.05, 0.07, gap], 0.03, [
    -0.68,
    -0.1,
    (DOOR_GAP[0] + DOOR_GAP[1]) / 2,
  ]);
  jukebox(k, music, !!upgrades.jukebox);
  for (const [x, z] of [
    [-0.65, 3.0],
    [W + 0.65, 3.0],
    [-0.66, FRONT_Z + 0.6],
    [W + 0.65, FRONT_Z + 0.6],
  ])
    planter(k, x, z);
  for (const x of [-0.58, W + 0.58]) {
    for (const z of [2.0, FRONT_Z + 0.26]) {
      k.cyl('paint', '#34474b', 0.047, 0.075, 3.65, [x, 1.585, z], { seg: 14 });
      k.cyl('chrome', '#8a9a99', 0.1, 0.14, 0.09, [x, -0.195, z], { seg: 14 });
      k.sphere('chrome', C.chrome, 0.073, [x, 3.43, z], { seg: 10 });
      k.box('chrome', '#859994', [0.17, 0.12, 0.08], [x, 3.3, z]);
    }
    festoon(k, lit, [x, 3.34, 2], [x, 3.34, FRONT_Z + 0.26], 0.44, 13);
    // Lower front string is pushed beyond the rail: its screen projection
    // stays below the seated guests and their bubbles, even at 36 degrees.
    k.cyl('paint', '#34474b', 0.045, 0.065, 2.79, [x, 1.155, FRONT_Z + 1.22], {
      seg: 12,
    });
    k.cyl('chrome', '#849c9b', 0.11, 0.14, 0.07, [x, -0.195, FRONT_Z + 1.22], {
      seg: 12,
    });
    k.sphere('chrome', C.chrome, 0.065, [x, 2.56, FRONT_Z + 1.22], { seg: 8 });
  }
  festoon(
    k,
    lit,
    [-0.58, 2.52, FRONT_Z + 1.22],
    [W + 0.58, 2.52, FRONT_Z + 1.22],
    0.35,
    17,
  );
  joinKit(group, k, materials, 'patio-static');
  joinKit(group, lit, own.materials, 'patio-bulbs');
  joinKit(group, music, musicOwn.materials, 'jukebox-tubes');
  const lights: THREE.PointLight[] = [];
  for (const [x, y, z] of [
    [2.3, 2.9, 0.9],
    [8.7, 2.9, 0.9],
    [5.5, 3.8, 7.3],
  ]) {
    const l = new THREE.PointLight(
      '#ffc390',
      quality === 'high' ? 7 : 4.5,
      12,
      2,
    );
    l.position.set(x, y, z);
    l.castShadow = false;
    lights.push(l);
    group.add(l);
  }
  batch(group);
  const disposeGeometry = modelDisposer(group, [
    own.glow,
    musicOwn.glow,
    tex,
    deckMat,
    pools.texture,
    pools.material,
  ]);
  let disposed = false;
  return {
    group,
    landmarks: [
      ...car.landmarks.map((p) => p.clone().add(car.group.position)),
      ...place.landmarks,
      new THREE.Vector3(-1.15, 2.02, 0.8),
      new THREE.Vector3(-1.1, 1.12, FRONT_Z + 0.74),
      new THREE.Vector3(W + 1.1, 1.12, FRONT_Z + 0.74),
      ...(upgrades.biggerPatio
        ? [new THREE.Vector3(W + 1.52, 0.06, 10.2)]
        : []),
      new THREE.Vector3(-0.68, 3.51, FRONT_Z + 0.26),
      new THREE.Vector3(W + 0.68, 3.51, FRONT_Z + 0.26),
    ],
    update(cues) {
      car.update(cues);
      truck.update(cues);
      place.update(cues);
      const level = lightLevel(cues);
      own.glow.color.setScalar(3 * level);
      pools.material.color.set('#ffc278').multiplyScalar(2.4 * level);
      musicOwn.glow.color.setScalar(
        2.6 *
          level *
          (cues.jukebox
            ? 1.15 + (cues.reduced ? 0 : Math.sin(cues.time * 3) * 0.12)
            : 0.35),
      );
      gate.rotation.y = -THREE.MathUtils.clamp(cues.gate, 0, 1) * 1.36;
      for (const light of lights)
        light.intensity = (quality === 'high' ? 7 : 4.5) * level;
    },
    dispose() {
      if (disposed) return;
      disposed = true;
      for (const child of [car, truck, place]) {
        child.group.removeFromParent();
        child.dispose();
      }
      disposeGeometry();
    },
  };
};
